rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isOwner(userId) {
      return request.auth != null && request.auth.uid == userId;
    }
    match /users/{userId} {
      allow read, write: if isOwner(userId);
    }
    match /mock_banks/{bankId}/accounts/{userId} {
      // Read stays owner-only. Write is opened to any signed-in user because a payment
      // needs to credit the RECEIVER's balance too, not just the sender's own account —
      // this is a mock/demo app with no real money, so this is fine here. If you ever
      // point this at anything real, move balance transfers into a Cloud Function
      // (Admin SDK) instead of relaxing this rule.
      allow read: if isOwner(userId);
      allow write: if request.auth != null;
    }
    match /transactions/{txnId} {
      allow read: if request.auth != null && (
        request.auth.uid == resource.data.senderUid ||
        request.auth.uid == resource.data.receiverUid
      );
      allow create: if request.auth != null
        && request.auth.uid == request.resource.data.senderUid;
      allow update, delete: if false;
    }
    match /counters/{counterId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null; // needed for atomic bank-assignment counter
    }
  }
}
